   179→function MedicineDialog({ open, onOpenChange, editing, cats, types, manuf, units }: any) {
   180→  const qc = useQueryClient()
   181→  const [form, setForm] = useState<any>(editing || { name: '', genericName: '', categoryId: '', typeId: '', manufacturerId: '', unitId: '', barcode: '', productCode: '', purchasePrice: 0, sellingPrice: 0, discountPercent: 0, vatPercent: 0, minStockLevel: 10, boxQuantity: 1, stripsPerBox: 0, tabletsPerStrip: 0, description: '', prescription: false, isActive: true })
   182→  const [saving, setSaving] = useState(false)
   183→
   184→  async function save() {
   185→    setSaving(true)
   186→    try {
   187→      const url = editing ? `/api/medicines/${editing.id}` : '/api/medicines'
   188→      const method = editing ? 'PUT' : 'POST'
   189→      const res = await fetch(url, { method, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(form) })
   190→      const d = await res.json()
   191→      if (!res.ok) throw new Error(d.error)
   192→      toast.success(editing ? 'Medicine updated' : 'Medicine added')
   193→      qc.invalidateQueries({ queryKey: ['medicines'] })
   194→      onOpenChange(false)
   195→    } catch (e: any) { toast.error(e.message) } finally { setSaving(false) }
   196→  }
   197→
   198→  return (
   199→    <Dialog open={open} onOpenChange={onOpenChange}>
   200→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto scroll-thin">
   201→        <DialogHeader><DialogTitle>{editing ? 'Edit Medicine' : 'Add Medicine'}</DialogTitle></DialogHeader>
   202→        <div className="grid grid-cols-1 md:grid-cols-2 gap-4 py-2">
   203→          <Field label="Brand Name *"><Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} /></Field>
   204→          <Field label="Generic Name"><Input value={form.genericName} onChange={(e) => setForm({ ...form, genericName: e.target.value })} /></Field>
   205→          <Field label="Category"><Select value={form.categoryId || 'none'} onValueChange={(v) => setForm({ ...form, categoryId: v === 'none' ? null : v })}><SelectTrigger><SelectValue placeholder="Select" /></SelectTrigger><SelectContent><SelectItem value="none">None</SelectItem>{cats?.map((c: any) => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}</SelectContent></Select></Field>
   206→          <Field label="Medicine Type"><Select value={form.typeId || 'none'} onValueChange={(v) => setForm({ ...form, typeId: v === 'none' ? null : v })}><SelectTrigger><SelectValue placeholder="Select" /></SelectTrigger><SelectContent><SelectItem value="none">None</SelectItem>{types?.map((t: any) => <SelectItem key={t.id} value={t.id}>{t.name}</SelectItem>)}</SelectContent></Select></Field>
   207→          <Field label="Manufacturer"><Select value={form.manufacturerId || 'none'} onValueChange={(v) => setForm({ ...form, manufacturerId: v === 'none' ? null : v })}><SelectTrigger><SelectValue placeholder="Select" /></SelectTrigger><SelectContent><SelectItem value="none">None</SelectItem>{manuf?.map((m: any) => <SelectItem key={m.id} value={m.id}>{m.name}</SelectItem>)}</SelectContent></Select></Field>
   208→          <Field label="Unit"><Select value={form.unitId || 'none'} onValueChange={(v) => setForm({ ...form, unitId: v === 'none' ? null : v })}><SelectTrigger><SelectValue placeholder="Select" /></SelectTrigger><SelectContent><SelectItem value="none">None</SelectItem>{units?.map((u: any) => <SelectItem key={u.id} value={u.id}>{u.name}</SelectItem>)}</SelectContent></Select></Field>
   209→          <Field label="Barcode"><Input value={form.barcode} onChange={(e) => setForm({ ...form, barcode: e.target.value })} /></Field>
   210→          <Field label="Product Code"><Input value={form.productCode} onChange={(e) => setForm({ ...form, productCode: e.target.value })} placeholder="Auto-generated if empty" /></Field>
   211→          <Field label="Purchase Price"><Input type="number" step="0.01" value={form.purchasePrice} onChange={(e) => setForm({ ...form, purchasePrice: Number(e.target.value) })} /></Field>
   212→          <Field label="Selling Price"><Input type="number" step="0.01" value={form.sellingPrice} onChange={(e) => setForm({ ...form, sellingPrice: Number(e.target.value) })} /></Field>
   213→          <Field label="Discount %"><Input type="number" step="0.01" value={form.discountPercent} onChange={(e) => setForm({ ...form, discountPercent: Number(e.target.value) })} /></Field>
   214→          <Field label="VAT %"><Input type="number" step="0.01" value={form.vatPercent} onChange={(e) => setForm({ ...form, vatPercent: Number(e.target.value) })} /></Field>
   215→          <Field label="Min Stock Level"><Input type="number" value={form.minStockLevel} onChange={(e) => setForm({ ...form, minStockLevel: Number(e.target.value) })} /></Field>
   216→          <Field label="Box Quantity"><Input type="number" value={form.boxQuantity} onChange={(e) => setForm({ ...form, boxQuantity: Number(e.target.value) })} /></Field>
   217→          <Field label="Strips per Box"><Input type="number" value={form.stripsPerBox} onChange={(e) => setForm({ ...form, stripsPerBox: Number(e.target.value) })} /></Field>
   218→          <Field label="Tablets per Strip"><Input type="number" value={form.tabletsPerStrip} onChange={(e) => setForm({ ...form, tabletsPerStrip: Number(e.target.value) })} /></Field>
   219→          <div className="md:col-span-2"><Field label="Description"><Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={2} /></Field></div>
   220→          <div className="flex items-center gap-6">
   221→            <label className="flex items-center gap-2 text-sm"><Switch checked={form.prescription} onCheckedChange={(v) => setForm({ ...form, prescription: v })} /> Prescription Required</label>
   222→            <label className="flex items-center gap-2 text-sm"><Switch checked={form.isActive} onCheckedChange={(v) => setForm({ ...form, isActive: v })} /> Active</label>
   223→          </div>
   224→        </div>
   225→        <DialogFooter>
   226→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   227→          <Button onClick={save} disabled={saving || !form.name} className="bg-primary">{saving ? 'Saving...' : 'Save'}</Button>
   228→        </DialogFooter>
   229→      </DialogContent>
   230→    </Dialog>

   263→function SimpleMasterList({ model, label, field, hasDesc }: { model: string; label: string; field: string; hasDesc?: boolean }) {
   264→  const qc = useQueryClient()
   265→  const { data, isLoading } = useQuery({ queryKey: [model], queryFn: async () => (await fetch(`/api/${model}`)).json() })
   266→  const [open, setOpen] = useState(false)
   267→  const [editing, setEditing] = useState<any>(null)
   268→  const [form, setForm] = useState<any>({ name: '', description: '' })
   269→  const [deleteId, setDeleteId] = useState<string | null>(null)
   270→
   271→  const saveMut = useMutation({
   272→    mutationFn: async (data: any) => {
   273→      const url = editing ? `/api/${model}/${editing.id}` : `/api/${model}`
   274→      const method = editing ? 'PUT' : 'POST'
   275→      const res = await fetch(url, { method, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   276→      const d = await res.json()
   277→      if (!res.ok) throw new Error(d.error)
   278→      return d
   279→    },
   280→    onSuccess: () => { toast.success('Saved'); qc.invalidateQueries({ queryKey: [model] }); setOpen(false) },
   281→    onError: (e: any) => toast.error(e.message),
   282→  })
   283→  const delMut = useMutation({
   284→    mutationFn: async (id: string) => { const res = await fetch(`/api/${model}/${id}`, { method: 'DELETE' }); const d = await res.json(); if (!res.ok) throw new Error(d.error); return d },
   285→    onSuccess: () => { toast.success('Deleted'); qc.invalidateQueries({ queryKey: [model] }) },
   286→    onError: (e: any) => toast.error(e.message),
   287→  })
   288→
   289→  const items = data?.items || []
   290→  return (
   291→    <Card className="border-slate-200 shadow-sm">
   292→      <div className="p-4 flex justify-between items-center border-b border-slate-100">
   293→        <h3 className="font-semibold text-slate-700">{label}s ({items.length})</h3>
   294→        <Button onClick={() => { setEditing(null); setForm({ name: '', description: '' }); setOpen(true) }} size="sm" className="bg-primary"><Plus className="w-4 h-4 mr-1" /> Add {label}</Button>
   295→      </div>
   296→      {isLoading ? <LoadingState /> : items.length === 0 ? <EmptyState title={`No ${label.toLowerCase()}s`} /> : (
   297→        <Table>
   298→          <TableHeader><TableRow className="bg-slate-50"><TableHead>Name</TableHead>{hasDesc && <TableHead>Description</TableHead>}<TableHead className="text-right">Actions</TableHead></TableRow></TableHeader>
   299→          <TableBody>
   300→            {items.map((it: any) => (
   301→              <TableRow key={it.id}>
   302→                <TableCell className="font-medium">{it.name}</TableCell>
   303→                {hasDesc && <TableCell className="text-sm text-slate-500">{it.description || '-'}</TableCell>}
   304→                <TableCell className="text-right">
   305→                  <Button variant="ghost" size="icon" onClick={() => { setEditing(it); setForm({ name: it.name, description: it.description || '' }); setOpen(true) }}><Edit className="w-4 h-4" /></Button>
   306→                  <Button variant="ghost" size="icon" onClick={() => setDeleteId(it.id)} className="text-destructive"><Trash2 className="w-4 h-4" /></Button>
   307→                </TableCell>
   308→              </TableRow>
   309→            ))}
   310→          </TableBody>
   311→        </Table>
   312→      )}
   313→      <Dialog open={open} onOpenChange={setOpen}>
   314→        <DialogContent>
   315→          <DialogHeader><DialogTitle>{editing ? `Edit ${label}` : `Add ${label}`}</DialogTitle></DialogHeader>
   316→          <div className="space-y-3 py-2">
   317→            <Field label="Name *"><Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} /></Field>
   318→            {hasDesc && <Field label="Description"><Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={2} /></Field>}
   319→          </div>
   320→          <DialogFooter><Button variant="outline" onClick={() => setOpen(false)}>Cancel</Button><Button onClick={() => saveMut.mutate(form)} disabled={!form.name} className="bg-primary">Save</Button></DialogFooter>

   322→      </Dialog>
